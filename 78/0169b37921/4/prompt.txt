## Key Patterns to Apply

### 1. Context + Custom Hook Pattern
```typescript
// context.tsx
const MyContext = createContext<ContextValue | null>(null);

export function MyProvider({ children, ...props }) {
  const state = useMyState(props);
  return <MyContext value={state}>{children}</MyContext>;
}

export function useMyContext() {
  const context = use(MyContext);
  if (!context) throw new Error('useMyContext must be used within MyProvider');
  return context;
}
```

### 2. Compound Component Pattern
```typescript
// index.tsx
export function Card({ children, className }) {
  return <div className={className}>{children}</div>;
}
Card.Header = CardHeader;
Card.Body = CardBody;
Card.Footer = CardFooter;
```

### 3. Hook Extraction Pattern
```typescript
// Component extracts business logic to hook
function MyForm(props) {
  const { form, isSubmitting, onSubmit } = useMyForm(props);
  return <form onSubmit={onSubmit}>...</form>;
}

// Hook contains all state management
function useMyForm({ onSuccess }) {
  const mutation = useMutation(...);
  const form = useForm(...);
  return { form, isSubmitting: mutation.isPending, onSubmit: form.handleSubmit };
}
```

### 4. Data-Driven UI Pattern
```typescript
// config.ts
export const items = [
  { id: 'a', label: 'Item A', icon: IconA },
  { id: 'b', label: 'Item B', icon: IconB },
];

// component.tsx
export function ItemList() {
  return items.map(item => <Item key={item.id} {...item} />);
}
```
