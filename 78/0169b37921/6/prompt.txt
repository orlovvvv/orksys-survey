export type MemberRole = "owner" | "admin" | "member";

## Key Patterns to Follow

### Form Pattern (from create-organization-form.tsx)
```typescript
const form = useForm({
  defaultValues: { email: "", role: "member" },
  onSubmit: async ({ value }) => {
    const result = await authClient.organization.inviteMember({
      email: value.email,
      role: value.role,
    });
    if (result.error) {
      toast.error(result.error.message);
      return;
    }
    toast.success("Invitation sent");
    onSuccess?.();
  },
  validators: {
    onSubmit: z.object({
      email: z.string().email(),
      role: z.enum(["member", "admin"]),
    }),
  },
});
```

### Responsive Dialog/Sheet Pattern
```typescript
function InviteMemberDialog() {
  const isMobile = useIsMobile();
  const [open, setOpen] = useState(false);

  if (isMobile) {
    return (
      <Sheet open={open} onOpenChange={setOpen}>
        <SheetTrigger asChild><Button>Invite</Button></SheetTrigger>
        <SheetContent><InviteForm onSuccess={() => setOpen(false)} /></SheetContent>
      </Sheet>
    );
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild><Button>Invite</Button></DialogTrigger>
      <DialogContent><InviteForm onSuccess={() => setOpen(false)} /></DialogContent>
    </Dialog>
  );
}
```

## API Methods (No new routes needed)

All operations use existing Better Auth client methods:

| Operation | Method |
|-----------|--------|
| List members | `authClient.organization.listMembers()` |
| Invite member | `authClient.organization.inviteMember({ email, role })` |
| Remove member | `authClient.organization.removeMember({ memberIdOrEmail })` |
| Update role | `authClient.organization.updateMemberRole({ memberId, role })` |
| List invitations | `authClient.organization.listInvitations()` |
| Cancel invitation | `authClient.organization.cancelInvitation({ invitationId })` |
| Get invitation | `authClient.organization.getInvitation({ id })` |
| Accept invitation | `authClient.organization.acceptInvitation({ invitationId })` |
| Reject invitation | `authClient.organization.rejectInvitation({ invitationId })` |

## Permission Rules

From `packages/auth/src/permissions.ts`:

- **Owner**: Full access, can assign admin role, manage billing
- **Admin**: Can invite/remove members, create/cancel invitations
- **Member**: Can view organization, cannot manage members

UI should hide action buttons for users without permission.
